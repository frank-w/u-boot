scriptaddr=0x83000000
bpi=bananapi
board=bpi-r2
service=linux
device=mmc
partition=1:1
bootenv=uEnv.txt
kernel=uImage
loadaddr=0x80200000
#default bootargs will be overidden by buildargs
bootargs=console=ttyS0,115200 root=/dev/mmcblk0p2 rw rootwait ip=dhcp

kernel414=uImage_4.14.91-main

console=earlyprintk console=ttyS0,115200
root=/dev/mmcblk0p2 rootfstype=ext4 rootwait
bootopts=vmalloc=496M debug=7 initcall_debug=0
graphic=video=1920x1080 console=tty1 fbcon=map:0 drm.debug=0x7

buildargs=env set bootargs "board=${board} ${console} root=${root} ${bootopts} ${graphic}"

checkenv=test -e ${device} ${partition} ${bpi}/${board}/${service}/${bootenv}
importenv=env import -t ${scriptaddr} ${filesize}
loadbootenv=if load ${device} ${partition} ${scriptaddr} ${bpi}/${board}/${service}/${bootenv};then run importenv;else echo "load (${bootenv}) failed";fi
reenv set=env default -a;env print;

checksd=info ${device} 1:1
selectmmc=if run checksd; then echo Boot from SD ; env set partition 1:1;else echo Boot from eMMC; env set partition 0:1 ; fi;

newboot=run buildargs;env print bootargs;load ${device} ${partition} ${loadaddr} ${bpi}/${board}/${service}/${kernel}; bootm

reloadmenu=run selectmmc;if run checkenv; then run loadbootenv; else echo file not found; fi;bootmenu;

lskernel=ls ${device} ${partition} ${bpi}/${board}/${service};
lsdtb=ls ${device} ${partition} ${bpi}/${board}/${service}/dtb
askkernel=askenv kernelinput "enter uImage-name:";
askdtb=askenv dtbinput "enter dtb-name:";
askuboot=askenv ubootinput "enter uboot-name:";

#bootmenu
boot0=run lskernel;run askkernel;if env print kernelinput ;then env set kernel ${kernelinput};run lsdtb;run askdtb;if env print dtbinput ;then env set fdt ${dtbinput};fi; run newboot2; fi;
boot1=run selectmmc; run loadbootenv; run importenv; run newboot;
boot14=env set kernel ${kernel414}; run newboot;
boot19=env set kernel ${kernel419}; run newboot;
boot19_dt=env set kernel ${kernel419_nodt}; env set fdt ${fdt419}; env set dtolist ${dtolist419}; run newboot2;

bootmenu_default=2

bootmenu_0=Enter kernel-name to boot from SD/EMMC.=run boot0
bootmenu_1=Boot kernel from TFTP.=run bootnet
bootmenu_2=Boot from SD/EMMC.=run boot1
bootmenu_3=Boot from SD/EMMC 4.14.=run boot14
bootmenu_4=Boot from SD/EMMC 4.19 (fdt/dto).=run boot19_dt

#automatic reload from sd/emmc
bootdelay=0
bootcmd=env set bootdelay 3; run reloadmenu;

png=ping ${serverip}

#Netboot
ipaddr=192.168.0.18
netmask=255.255.255.0
serverip=192.168.0.11

bootfile=uImage
ubootfile=u-boot.bin
netbootargs=root=/dev/mmcblk0p2 rw rootwait
#ip=dhcp sets ip on eth0 instead of wan

lstftp=if tftp ${loadaddr} ${serverip}:files.lst;then setexpr listend ${loadaddr} + ${filesize};mw.b ${listend} 00 2;strings ${loadaddr};else echo "tftp download failed";fi
lstftpu=if tftp ${loadaddr} ${serverip}:bootfiles.lst;then setexpr listend ${loadaddr} + ${filesize};mw.b ${listend} 00 2;strings ${loadaddr};else echo "tftp download failed";fi
#md.b ${loadaddr} 60;

ubootaddr=0x81E00000
bootnet=run lstftp;run askkernel;if env print kernelinput ;then env set bootfile "${kernelinput}"; fi;env print; env set bootargs "${netbootargs} ${console} ${graphic}";if tftp ${loadaddr} ${bootfile};then bootm;fi
ubootnet=run lstftpu;run askuboot;if env print ubootinput ;then env set ubootfile "${ubootinput}"; fi; if tftp ${ubootaddr} ${ubootfile};then icache off;dcache off;go ${ubootaddr};fi

#NFS
nfsip=192.168.0.11
nfsrfs=/var/nfs/bpi-r2
setnfsargs=env set netbootargs "root=/dev/nfs rw nfsroot=${nfsip}:${nfsrfs},nfsvers=3,tcp nfsrootdebug";
bootnfs=run setnfsargs bootnet

#separate fdt+dto
dtaddr=0x83f00000
fdt_high=0xffffffff

kernel419_nodt=uImage_4.19.13-main_nodt
fdt419=4.19.13-main.dtb
dtolist419=bpi-r2-mac.dtb

loadfdt=load ${device} ${partition} ${dtaddr} ${bpi}/${board}/${service}/dtb/${fdt}
loadkernel=echo "loading kernel ${kernel}...";load ${device} ${partition} ${loadaddr} ${bpi}/${board}/${service}/${kernel}

loaddto=echo "loaddto:${dto}";fdt addr ${dtaddr};fdt resize 8192; setexpr fdtovaddr ${dtaddr} + F000;load ${device} ${partition} ${fdtovaddr} ${bpi}/${board}/${service}/dtb/${dto} && fdt apply ${fdtovaddr}
loaddtolist=for dto in ${dtolist} ; do run loaddto ; done

bootall=if env print fdt; then run loadfdt; if env print dtolist;then run loaddtolist;fi; bootm ${loadaddr} - ${dtaddr} ;else bootm; fi
newboot2=run buildargs; env print bootargs; run loadkernel; run bootall;

#sata/ahci
initsata=pci enum;scsi scan;

partsata=0:1
dirsata=EFI/ubuntu
filesata=test.txt
kernelsata=uImage_4.14.117-main

loadfilesata=load scsi ${partsata} ${scriptaddr} ${dirsata}/${filesata};setexpr listend ${scriptaddr} + ${filesize};mw.b ${listend} 00 2;strings ${scriptaddr};
loadkernelsata=load scsi ${partsata} ${loadaddr} ${dirsata}/${kernelsata};bootm
lssata=ls scsi ${partsata} ${$dirsata}

askaddr=askenv addrinput "enter address:";
lsmem=run askaddr;if env print addrinput;then env set baseaddr ${addrinput};run listmem; fi
listmem=if env print baseaddr;then env set i 0;while test ${i} -lt 10;do env set j 0;while test ${j} -lt 10;do env set offset "${i}${j}00";setexpr addr $baseaddr + ${offset};md $addr;setexpr j ${j} + 1; done;setexpr i ${i} + 1;done;fi
