board=bpi-r4
device=mmc
partition=0:5
bootenv=uEnv.txt
bootdevice=sd

fit=bpi-r4.itb
#bootconf="#conf-2"
#initrd=rootfs.cpio.zst

loadaddr=0x46000000
uaddr=0x41e00000
kaddr=0x46000000
rdaddr=0x48000000

root=/dev/mmcblk0p6 rootfstype=ext4 rootwait
console=earlycon=uart8250,mmio32,0x11000000
bootopts=debug=7

#version=-v0
wifi_en=0
sata_en=0

usb_part=0:1

nvme_boot=0:1
nvme_root=/dev/nvme0n1p2

ubi_part=0:3
ubi_partname=rootfs

#netboot
ipaddr=192.168.0.19
netmask=255.255.255.0
serverip=192.168.0.11

lstftp=if tftp ${loadaddr} ${serverip}:r4-itbfiles.lst;then setexpr listend ${loadaddr} + ${filesize};mw.b ${listend} 00 2;strings ${loadaddr};else echo "tftp download failed";fi
askkernel=askenv kernelinput "enter kernel-name:";
askbootnetfit=run lstftp askkernel;if env print kernelinput; then env set bootfile "${kernelinput}"; run bootnetfit;fi

askbootnvme=run usenvme; if test "$device" = "nvme"; then run askkernel;if env print kernelinput; then env set fit $kernelinput;run newboot;fi;fi
bootnvme=run usenvme; if test "$device" = "nvme"; then run loadenv; env print; run newboot;fi
bootubi=run useubi; if test "$device" = "ubi"; then run loadenv; env print; run newboot;fi

bootfile=bpi-r4.itb

checkenv=test -e ${device} ${partition} ${bootenv}
loadenvfile=if load ${device} ${partition} ${loadaddr} ${bootenv};then env import -t ${loadaddr} ${filesize};else echo "load (${bootenv}) failed";fi
loadenv=if run checkenv;then run loadenvfile;fi
reenv set=env default -a;env print;

loadkernel=load $device $partition $kaddr ${kernel}
buildargs=env set bootargs "board=${board} ${console} root=${root} ${bootopts}"

useusb=if usb info; then usb reset;else usb start;fi; if ls usb $usb_part $usb_path; then env set device usb;env set partition $usb_part;fi
usenvme=pci enum; nvme scan;nvme info; if ls nvme $nvme_boot;then env set device "nvme";env set partition $nvme_boot;env set root $nvme_root;else echo "nvme partition not found";fi
useubi=ubi part ubi;ubifsmount ubi0:$ubi_partname;if ls ubi $ubi_part;then env set device "ubi";env set partition $ubi_part;else echo "ubi partition $ubi_part not found";fi
umountubi=if test $device == "ubi";then ubifsumount;fi

useinitrd=env set bootopts "${bootopts} initrd=${rdaddr},100M";
checkrd=if env print initrd;then if load ${device} ${partition} ${rdaddr} ${initrd};then run useinitrd;fi;fi

newboot=run checkrd; run setbootconf; run buildargs;if env print fit;then env set kernel ${fit};if run loadkernel; then run umountubi; bootm ${kaddr}${bootconf}; fi;fi;
bootnetfit=run checkrd; run setbootconf;run buildargs;if tftp $kaddr ${bootfile}; then bootm ${kaddr}${bootconf};fi

checkmmc=if mmc partconf 0; then echo "emmc available";mmcdev="emmc";else echo "sd available";mmcdev="sd";fi;env set ovmmc "#ov-$mmcdev";
checkspi=if sf probe; then echo "NOR available";spidev="nor";else echo "NAND available";spidev="nand";fi
checksata=if test -n "$sata_en" && test $sata_en -eq 1;then env set addconf "$addconf#sata"; fi
checkwifi=if test -n "$wifi_en" && test $wifi_en -eq 1;then env set addconf "$addconf#wifi"; fi
check2g5=if test -n "$is2g5" && test $is2g5 -eq 1;then env set variant "-2g5"; fi
checkr4provariant=if test -n "$isr4pro4e" && test $isr4pro4e -eq 1;then env set variant "-r4pro-4e";fi
checkr4pro=if test -n "$isr4pro" && test $isr4pro -eq 1;then env set variant "-r4pro";run checkr4provariant; env set ovmmc "#ov-pro-$mmcdev"; fi
#r4lite needs different mmc-overlays based on shematic
checkr4lite=if test -n "$isr4lite" && test $isr4lite -eq 1;then env set variant "-r4lite";env set ovmmc "#ov-lite-$mmcdev";run checkspi;env set ovspi "#ov-lite-$spidev";fi
setbootconf=run check2g5;run checkmmc;run checksata;run checkwifi;run checkr4lite;run checkr4pro;env set bootconf "#conf-base$variant$version$ovmmc$ovspi$addconf";env print bootconf

wrspimnand=if env print bl2file;then if load $device $partition $loadaddr $bl2file;then mtd erase spi-nand0 0x0 0x100000;mtd write spi-nand0 $loadaddr 0x0 0x100000;else echo "loading bl2 $bl2file failed";fi;else echo "bl2file not defined";fi;if env print fipfile;then if load $device $partition $loadaddr $fipfile;then mtd erase spi-nand0 0x580000 0x200000;mtd write spi-nand0 $loadaddr 0x580000 0x200000;else echo "loading fip $fipfile failed";fi;else echo "fipfile not defined";fi
wremmc=if env print bl2file;then mmc partconf 0 1 1 1;if load $device $partition $loadaddr $bl2file;then mmc erase 0x0 0x400;mmc write ${loadaddr} 0x0 0x400;else echo "loading bl2 $bl2file failed";fi;else echo "bl2file not defined";fi;mmc partconf 0 1 1 0;if env print fipfile;then if load $device $partition $loadaddr $fipfile;then mmc erase 0x3400 0x1000;mmc write ${loadaddr} 0x3400 0x1000;else echo "loading fip $fipfile failed";fi;else echo "fipfile not defined";fi

bootdelay=0
bootcmd=env set bootdelay 3; run loadenv;bootmenu;

lskernel=ls ${device} ${partition};
askkernel=askenv kernelinput "enter kernel name:";

boot0=run lskernel;run askkernel;if env print kernelinput ;then env set fit ${kernelinput}; run newboot; fi;

bootmenu_0=Enter kernel-name to boot from SD/EMMC.=run boot0
bootmenu_1=Boot kernel from TFTP.=run askbootnetfit
bootmenu_2=Boot from SD/EMMC.=run newboot
bootmenu_3=Boot kernel from NVME.=run bootnvme
bootmenu_4=Boot kernel from UBI.=run bootubi

bootmenu_default=2
